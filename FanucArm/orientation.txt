    0.0,
    -0.70710678,
    0.0,
    0.70710678
])

        max_velocities=[

        ],

        ],

        active_joints=[
            "J1","J2","J3",
            "J4","J5","J6"
        ]
    )

    print("Duration =", jt.duration)

    dt = 1.0/60.0
    t = 0.0

    while t <= jt.duration:

        state = jt.get_target_state(t)

        robot.apply_action(
            ArticulationAction(
                joint_positions=state.joints.positions
            )
        )

        app.update()
        time.sleep(dt)

        t += dt

    print("Finished")
